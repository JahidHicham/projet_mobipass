SELECT
  NON EMPTY [Temps].[H_MOIS].[Mois].Members                    ON COLUMNS,
  NON EMPTY [Employé].[H_Type_Societe].[Société].Members       ON ROWS
FROM [Locations]
WHERE ([Measures].[Montant payé]);

-- b) Montants reellement payes et durees par semaine et par salarie (3 annees completes)
--    (3 000 salaries x 156 semaines : le NON EMPTY est indispensable)

SELECT
  {[Measures].[Montant payé], [Measures].[Durée]} ON COLUMNS,
  NonEmpty(
    CROSSJOIN(
      Descendants({[Temps].[H_SEM].[Année ISO].&[2023],
                   [Temps].[H_SEM].[Année ISO].&[2024],
                   [Temps].[H_SEM].[Année ISO].&[2025]},
                  [Temps].[H_SEM].[Semaine]),
      [Employé].[H_Type_Societe].[Employé].Members),
    [Measures].[Montant payé])                     ON ROWS
FROM [Locations];

/* =====================================================================
   S1 - SATURATION SAISONNIERE
   Question : quand et ou les employes ne trouvent-ils pas de produit a louer ?
   Indicateur : [Taux de demandes non satisfaites] (fait Réservation, NonHonoree = 1)
   ===================================================================== */

-- S1.1 Vue annuelle (ROLL-UP au niveau Année)
-- Question : la part de demandes non satisfaites est-elle stable d'une annee a l'autre ?
SELECT
  {[Measures].[Nombre de locations], [Measures].[Réservations non honorées],
   [Measures].[Taux de demandes non satisfaites]}              ON COLUMNS,
  [Temps].[H_MOIS].[Année].Members                             ON ROWS
FROM [Locations];

-- S1.2 DRILL-DOWN par mois sur l'annee retenue
-- Question : quels mois concentrent les demandes non satisfaites ?
SELECT
  {[Measures].[Nombre de locations], [Measures].[Réservations non honorées],
   [Measures].[Taux de demandes non satisfaites]}              ON COLUMNS,
  Descendants([Temps].[H_MOIS].[Année].&[2025],
              [Temps].[H_MOIS].[Mois])                         ON ROWS
FROM [Locations];

-- S1.3 DRILL-DOWN par semaine, classees : les semaines du pic
-- Question : quelles semaines precises sont les plus touchees ? (attendu : semaines 38-39)
SELECT
  {[Measures].[Réservations non honorées],
   [Measures].[Taux de demandes non satisfaites]}              ON COLUMNS,
  TopCount(
    Descendants([Temps].[H_SEM].[Année ISO].&[2025],
                [Temps].[H_SEM].[Semaine]),
    10, [Measures].[Réservations non honorées])                ON ROWS
FROM [Locations];

-- S1.4 SLICE sur la saison (ete 2025) et eclatement par secteur
-- Question : quels secteurs sont les plus touches pendant la saison chargee ?
SELECT
  {[Measures].[Réservations non honorées],
   [Measures].[Taux de demandes non satisfaites]}              ON COLUMNS,
  Order([Exemplaire].[H_GAMME].[Secteur].Members,
        [Measures].[Réservations non honorées], BDESC)         ON ROWS
FROM [Locations]
WHERE ([Temps].[Saison].&[été], [Temps].[H_MOIS].[Année].&[2025]);

-- S1.5 DRILL-DOWN secteur -> gamme -> produit
-- Question : quelles gammes et quels produits cristallisent la saturation ?
SELECT
  {[Measures].[Réservations non honorées],
   [Measures].[Taux de demandes non satisfaites]}              ON COLUMNS,
  TopCount(
    Descendants([Exemplaire].[H_GAMME].[Secteur].&[Vélos],
                [Exemplaire].[H_GAMME].[Produit]),
    10, [Measures].[Réservations non honorées])                ON ROWS
FROM [Locations]
WHERE ([Temps].[Saison].&[été], [Temps].[H_MOIS].[Année].&[2025]);

-- S1.6 ECLATEMENT par region (France) pour la gamme retenue
-- Question : dans quelles regions les demandes de cette gamme ne sont-elles pas satisfaites ?
SELECT
  {[Measures].[Réservations non honorées],
   [Measures].[Taux de demandes non satisfaites]}              ON COLUMNS,
  Order(Descendants([Employé].[H_France].[Pays].&[FR],
                    [Employé].[H_France].[Région]),
        [Measures].[Réservations non honorées], BDESC)         ON ROWS
FROM [Locations]
WHERE ([Exemplaire].[H_GAMME].[Gamme].[Vélos électriques urbains],
       [Temps].[Saison].&[été], [Temps].[H_MOIS].[Année].&[2025]);

-- S1.7 DRILL-DOWN geographique : de la region retenue vers ses departements
SELECT
  {[Measures].[Réservations non honorées],
   [Measures].[Taux de demandes non satisfaites]}              ON COLUMNS,
  Order(Descendants([Employé].[H_France].[Région].[Île-de-France],
                    [Employé].[H_France].[Département]),
        [Measures].[Réservations non honorées], BDESC)         ON ROWS
FROM [Locations]
WHERE ([Exemplaire].[H_GAMME].[Gamme].[Vélos électriques urbains],
       [Temps].[Saison].&[été], [Temps].[H_MOIS].[Année].&[2025]);

/* =====================================================================
   S2 - PROFILS DES LOCATAIRES
   Question : quels profils louent le plus, le plus longtemps, avec quelle remise ?
   profil = Catégorie ; classe de trajet = Super Catégorie ; domaine = Domaine / Division NAF
   (les mesures calculees sont celles du cube : plus de WITH MEMBER, sauf S2.6)
   ===================================================================== */

-- S2.1 Vue par classe de trajet
SELECT
  {[Measures].[Nombre de locations], [Measures].[Durée],
   [Measures].[Durée moyenne], [Measures].[Taux de remise effectif]}   ON COLUMNS,
  [Employé].[H_Categorie].[Super Catégorie].Members                    ON ROWS
FROM [Locations];

-- S2.2 DRILL-DOWN : profils (Catégorie) de la classe qui ressort
SELECT
  {[Measures].[Nombre de locations], [Measures].[Durée moyenne],
   [Measures].[Taux de remise effectif]}                               ON COLUMNS,
  Descendants([Employé].[H_Categorie].[Super Catégorie].&[Trajet long (plus de 15 km)],
              [Employé].[H_Categorie].[Catégorie])                     ON ROWS
FROM [Locations];

-- S2.3 ROLL-UP par domaine d'activite (division NAF) pour la classe retenue
SELECT
  {[Measures].[Nombre de locations], [Measures].[Durée]}               ON COLUMNS,
  NON EMPTY Order([Employé].[H_Domaine].[Division].Members,
                  [Measures].[Durée], BDESC)                           ON ROWS
FROM [Locations]
WHERE ([Employé].[H_Categorie].[Super Catégorie].&[Trajet long (plus de 15 km)]);

-- S2.4 Montant public / montant paye : classe de trajet x secteur (CROSSJOIN)
SELECT
  {[Measures].[Montant public], [Measures].[Montant payé],
   [Measures].[Taux de remise effectif]}                               ON COLUMNS,
  [Employé].[H_Categorie].[Super Catégorie].Members
    * [Exemplaire].[H_GAMME].[Secteur].Members                         ON ROWS
FROM [Locations];

-- S2.5 FILTER : profils dont le taux de remise effectif depasse un seuil
SELECT
  {[Measures].[Montant public], [Measures].[Montant payé],
   [Measures].[Taux de remise effectif]}                               ON COLUMNS,
  Filter([Employé].[H_Categorie].[Catégorie].Members,
         [Measures].[Taux de remise effectif] > 0.15)                  ON ROWS   -- seuil a ajuster d'apres S2.4
FROM [Locations];

-- S2.6 Part de chaque profil dans sa classe (pourcentage par rapport au parent)
WITH
  MEMBER [Measures].[Part de la classe] AS
    IIF(([Measures].[Durée], [Employé].[H_Categorie].CurrentMember.Parent) = 0, NULL,
        [Measures].[Durée] /
        ([Measures].[Durée], [Employé].[H_Categorie].CurrentMember.Parent)),
    FORMAT_STRING = '0.00%'
SELECT
  {[Measures].[Durée], [Measures].[Part de la classe]}                 ON COLUMNS,
  Order([Employé].[H_Categorie].[Catégorie].Members,
        [Measures].[Part de la classe], BDESC)                         ON ROWS
FROM [Locations];

-- S3.1 Poids de chaque gamme dans le secteur Vélos (pourcentage par rapport au parent)
WITH
  MEMBER [Measures].[Part du secteur] AS
    IIF(([Measures].[Nombre de locations], [Exemplaire].[H_GAMME].CurrentMember.Parent) = 0, NULL,
        [Measures].[Nombre de locations] /
        ([Measures].[Nombre de locations], [Exemplaire].[H_GAMME].CurrentMember.Parent)),
    FORMAT_STRING = '0.0%'
SELECT
  {[Measures].[Nombre de locations], [Measures].[Part du secteur]}     ON COLUMNS,
  Descendants([Exemplaire].[H_GAMME].[Secteur].&[Vélos],
              [Exemplaire].[H_GAMME].[Gamme])                          ON ROWS
FROM [Locations];

-- S3.2 Gamme x region (CROSSJOIN) : une gamme ressort-elle dans une region ?
SELECT
  [Exemplaire].[H_GAMME].[Gamme].Members                               ON COLUMNS,
  NON EMPTY Descendants([Employé].[H_France].[Pays].&[FR],
                        [Employé].[H_France].[Région])                 ON ROWS
FROM [Locations]
WHERE ([Measures].[Nombre de locations]);
